let current = 0;
let score = {s1:0,s2:0,s3:0,s4:0,s5:0,s6:0,s7:0,s8:0,s9:0,s10:0};

function startQuiz(){
  document.getElementById("home").style.display="none";
  document.getElementById("quiz").style.display="block";
  showQ();
}
function showQ(){
  const q = questions[current];
  document.getElementById("progress").innerText = `第${current+1}题 / 共${questions.length}题`;
  document.getElementById("qTitle").innerText = q.title;
  const box = document.getElementById("optsBox");
  box.innerHTML = "";
  q.opts.forEach((opt,idx)=>{
    const b = document.createElement("button");
    b.className = "opt";
    b.innerText = opt.text;
    b.onclick = ()=>{
      score[opt.k]++;
      current++;
      if(current >= questions.length){
        showResult();
      }else{
        showQ();
      }
    }
    box.appendChild(b);
  })
}
function showResult(){
  document.getElementById("quiz").style.display="none";
  document.getElementById("result").style.display="block";
  //找最高分
  let maxk = "s1";
  let maxv = 0;
  for(let k in score){
    if(score[k]>maxv){
      maxv=score[k];maxk=k;
    }
  }
  const r = resultText[maxk];
  document.getElementById("resTitle").innerText = r.title;
  document.getElementById("resDesc").innerText = r.desc;
}
function resetAll(){
  current=0;
  score={s1:0,s2:0,s3:0,s4:0,s5:0,s6:0,s7:0,s8:0,s9:0,s10:0};
  document.getElementById("result").style.display="none";
  document.getElementById("home").style.display="block";
}
